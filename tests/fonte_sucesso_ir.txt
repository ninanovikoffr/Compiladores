int $soma(int $a, int $b) {
    return $a + $b;
}

int $maior(int $a, int $b) {
    if ($a > $b)
        return $a;
    else
        return $b;
}

int $x = 0;
int $y = 3;
int $z;
float $f = 2.5;

read($x);

$z = $soma($x, $y);
print($z);

if (($z > 10) && ($x != 0)) {
    print("maior");
} else {
    print("menor");
}

while ($x < 5) {
    $x = $x + 1;
    $z = $z + $x;
}

$z = $z % 2;
$z = -$z;
$z = !($z == 0);

$f = $f + $x * 2;

$z = $maior($z, $x);

print($x);
print($z);
print($f);
